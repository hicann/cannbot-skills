# Ascend C 源码同库编译变体；配套src/proto_reg.cpp和src/proto_reg.h。
# 工具链与部署条件见 references/language-launch.md、build-verify.md。
cmake_minimum_required(VERSION 3.16)
if(NOT DEFINED ASCEND_HOME_PATH)
    set(ASCEND_HOME_PATH "$ENV{ASCEND_HOME_PATH}")
endif()
if(NOT ASCEND_HOME_PATH)
    message(FATAL_ERROR "ASCEND_HOME_PATH is required")
endif()
# ASC语言模块须在project启用ASC之前加载，CANN环境提供其搜索路径。
find_package(ASC REQUIRED)
project(my_custom_op LANGUAGES ASC CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ASCEND_NPU_ARCH "" CACHE STRING "Target compiler architecture, e.g. dav-2201")
if(NOT ASCEND_NPU_ARCH)
    message(FATAL_ERROR "Set ASCEND_NPU_ARCH for the target compiler")
endif()
add_library(cust_opapi SHARED
    add_custom_kernel/add_custom.asc src/add_custom_op.cpp src/proto_reg.cpp)
target_include_directories(cust_opapi PRIVATE
    "${ASCEND_HOME_PATH}/include" "${ASCEND_HOME_PATH}/include/external"
    "${ASCEND_HOME_PATH}/include/graph_metadef/external"
    "${CMAKE_CURRENT_SOURCE_DIR}/src" "${CMAKE_CURRENT_SOURCE_DIR}/add_custom_kernel")
target_link_directories(cust_opapi PRIVATE "${ASCEND_HOME_PATH}/lib64")
target_compile_definitions(cust_opapi PRIVATE __ASCEND_C__ _GLIBCXX_USE_CXX11_ABI=0)
target_compile_options(cust_opapi PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=${ASCEND_NPU_ARCH}>)
# custom_op_registry_static 是静态库 libcustom_op_registry_static.a（无同名 .so），提供离线 OM
# 加载必需的 pull creator 符号（GetRegisteredCustomOpCreator*）；删除后 OM 加载整体失败。
target_link_libraries(cust_opapi PRIVATE
    -Wl,--no-as-needed ascendcl ascendc_runtime acl_rt ge_common ge_compiler ge_executor
    runtime register gert custom_op_registry_static lowering dl -Wl,--as-needed)
string(TOLOWER "${CMAKE_SYSTEM_NAME}" CUSTOM_OP_OS_DIR)
string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" CUSTOM_OP_ARCH_DIR)
if(CUSTOM_OP_ARCH_DIR STREQUAL "arm64")
    set(CUSTOM_OP_ARCH_DIR "aarch64")
elseif(CUSTOM_OP_ARCH_DIR STREQUAL "amd64")
    set(CUSTOM_OP_ARCH_DIR "x86_64")
endif()
set(OPP_OUTPUT_PATH "${CMAKE_BINARY_DIR}/opp")
set_target_properties(cust_opapi PROPERTIES OUTPUT_NAME "cust_opapi"
    LIBRARY_OUTPUT_DIRECTORY "${OPP_OUTPUT_PATH}/op_graph/lib/${CUSTOM_OP_OS_DIR}/${CUSTOM_OP_ARCH_DIR}")
add_custom_command(TARGET cust_opapi POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "${OPP_OUTPUT_PATH}/op_graph/include"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/src/proto_reg.h"
            "${OPP_OUTPUT_PATH}/op_graph/include/proto_reg.h")
# 已直接写入OPP目录，不再install到相同位置。
